📖 SDK Quickstart & Integration 

// 1. Include the CrimX SDK in your web app:
import { CrimX } from "https://crimsonflame.net/crimx.js";

// 2. Redirect player to CrimX Auth:
const loginUrl = CrimX.getAuthUrl({
  clientId: "YOUR_CLIENT_ID",
  redirectUri: "https://yourgame.com/callback",
  scope: "profile email",
  state: "secure_random_state_123"
});
window.location.href = loginUrl;

// 3. On your callback page, parse response:
const callback = CrimX.parseCallback();
if (callback.success) {
  // Exchange code for verified player profile!
  const session = await CrimX.exchangeCode({
    code: callback.code,
    clientId: "YOUR_CLIENT_ID"
  });
  console.log("Logged in as:", session.user.name, session.user.uid);
} else {
  // Handles errors like user cancelling or access denied
  alert("In-Game Login Error: " + callback.errorDescription);
}
